BLEX Developer API

{% translate "Developer API quickstart" %}

{% translate "The generated API documentation covers every endpoint. This guide walks through the shortest path from creating a token to publishing a Markdown draft." %}

{% translate "Open API documentation" %} {% translate "Manage tokens" %}

{% translate "1. Create a token" %}

  1. {% translate "Log in to BLEX with an author account." %}
  2. {% blocktranslate trimmed %} Open /settings/developer-api. {% endblocktranslate %}
  3. {% blocktranslate trimmed %} Select both Read posts and Write posts to create and publish posts. {% endblocktranslate %}
  4. {% translate "Copy the full token shown immediately after creation. It cannot be viewed again." %}
export BLEX_ORIGIN="{{ site_origin }}"
export BLEX_TOKEN="blex_pat_..."

{% translate "2. Verify the token" %}

curl "$BLEX_ORIGIN/api/developer/v1/me" \
  -H "Authorization: Bearer $BLEX_TOKEN"

{% translate "3. Create a Markdown draft" %}

curl -X POST "$BLEX_ORIGIN/api/developer/v1/posts" \
  -H "Authorization: Bearer $BLEX_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "status": "draft",
    "title": "{{ example_post_title }}",
    "markdown": "# Hello BLEX\n\n{{ example_draft_body }}",
    "tags": ["api", "automation"],
    "slug": "api-first-draft"
  }'

{% blocktranslate trimmed %} Keep the data.id and data.updated_at values from the response. Replace 123 in the following examples with the returned data.id. {% endblocktranslate %} 

{
  "data": {
    "id": 123,
    "updated_at": "2026-05-30T00:00:00+00:00"
  }
}

{% translate "4. Upload an image" %}

curl -X POST "$BLEX_ORIGIN/api/developer/v1/images" \
  -H "Authorization: Bearer $BLEX_TOKEN" \
  -F "image=@./cover.png"

{% blocktranslate trimmed %} Use data.url from the response as the Markdown image URL. {% endblocktranslate %}

{% translate "5. Update the draft" %}

{% blocktranslate trimmed %} To reduce concurrent edit conflicts, send the updated_at value from a detail or creation response as expected_updated_at. {% endblocktranslate %} 

curl -X PATCH "$BLEX_ORIGIN/api/developer/v1/posts/123" \
  -H "Authorization: Bearer $BLEX_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "expected_updated_at": "2026-05-30T00:00:00+00:00",
    "description": "{{ example_post_description }}",
    "tags": ["api", "automation", "blex"]
  }'

{% translate "6. Publish the draft" %}

curl -X POST "$BLEX_ORIGIN/api/developer/v1/posts/123/publish" \
  -H "Authorization: Bearer $BLEX_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{}'
